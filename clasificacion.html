function renderizarParticipacionMes() { const anio = document.getElementById("filtro-anio").value; const mes = document.getElementById("filtro-mes").value; const nombreMes = NOMBRES_MES[mes] || ""; document.getElementById("titulo-participacion-mes").textContent = `📊 Actividad y Participación — ${nombreMes} ${anio}`; // 1. Calcular partidas del mes seleccionado const partidasMes = PARTIDAS.filter(p => p.anio == anio && String(p.mes).padStart(2, "0") === mes); const conteoJugadores = {}; partidasMes.forEach(p => { (p.jugadores || []).forEach(j => { const nombreOficial = resolverNombreOficial(j.nombre, EQUIVALENCIAS); conteoJugadores[nombreOficial] = (conteoJugadores[nombreOficial] || 0) + 1; }); }); // 2. RESPALDO HISTÓRICO: Si el mes actual tiene poca o nula actividad, completamos con el acumulado general del año if (Object.keys(conteoJugadores).length < 5) { PARTIDAS.filter(p => p.anio == anio).forEach(p => { (p.jugadores || []).forEach(j => { const nombreOficial = resolverNombreOficial(j.nombre, equivalencias); // Les damos un peso base menor o sumamos su histórico para mantenerlos visibles if (!conteoJugadores[nombreOficial]) { conteoJugadores[nombreOficial] = 0.1; // Marcador histórico para que aparezcan al inicio de mes } }); }); } const listaMiembros = []; const listaParticipantes = []; const listaInvitados = []; Object.entries(conteoJugadores).forEach(([nombre, totalPartidas]) => { const info = JUGADORES_INFO[nombre] || {}; const pais = info.pais ? info.pais.toLowerCase() : null; const item = { nombre, partidas: Math.round(totalPartidas), pais }; if (totalPartidas >= 5) { listaMiembros.push(item); } else if (totalPartidas >= 2) { listaParticipantes.push(item); } else { listaInvitados.push(item); } }); listaMiembros.sort((a, b) => b.partidas - a.partidas); listaParticipantes.sort((a, b) => b.partidas - a.partidas); listaInvitados.sort((a, b) => b.partidas - a.partidas); pintarColumnaMiniParticipacion("lista-miembros-comunidad", listaMiembros); pintarColumnaMiniParticipacion("lista-participantes-liga", listaParticipantes); pintarColumnaMiniParticipacion("lista-invitados-liga", listaInvitados); }
